                emergencyAbandonBtn: document.getElementById('emergency-abandon-btn'),
                emergencyPauseBtn: document.getElementById('emergency-pause-btn'),

                pausedTitle: document.getElementById('paused-title'),

            };

        // Bind all event listeners
        bindEvents() {
            Object.values(this.ui.inputs).forEach(input => {
                if (input) {
                    const eventType = (input.type === 'checkbox' || input.tagName === 'SELECT') ? 'change' : 'input';
                    if (!input.id.startsWith('live-')) { // Don't attach to live settings inputs yet
                        input.addEventListener(eventType, () => this.updateSettings());
                    }
                }
            });
            this.ui.darkModeToggle.addEventListener('change', () => this.toggleDarkMode());
            this.ui.tooltipToggle.addEventListener('change', (e) => {
                this.state.settings.showTooltips = e.target.checked;
                this.toggleAllTooltips(this.state.settings.showTooltips);
                this.saveState();
            });
            
            document.addEventListener('keydown', (e) => this.handleModalFocus(e));
        },
        
        initSortableSetup() {
            if (this.sortable.setup) this.sortable.setup.destroy();
            this.sortable.setup = new Sortable(this.ui.examList, {
                animation: 150, ghostClass: 'sortable-ghost',
                onEnd: (evt) => {
                    const movedExam = this.state.exams.splice(evt.oldIndex, 1)[0];
                    this.state.exams.splice(evt.newIndex, 0, movedExam);
                    this.saveState();
                },
            });
        },
        
        initSortableExamPage() {
            if (this.sortable.examPage) this.sortable.examPage.destroy();
            this.sortable.examPage = new Sortable(this.ui.examTimersGrid, {
                animation: 150, ghostClass: 'sortable-ghost',
                onEnd: (evt) => {
                    const movedExam = this.state.exams.splice(evt.oldIndex, 1)[0];
                    this.state.exams.splice(evt.newIndex, 0, movedExam);
                    this.saveState();
                    this.prepareExamPage();
                },
            });
        },
        
        saveState() {
            localStorage.setItem('examTimerState', JSON.stringify(this.state));
        },

        loadState() {
            const savedState = localStorage.getItem('examTimerState');
            if (savedState) {
                const loaded = JSON.parse(savedState);
                const settings = { ...this.defaultSettings, ...loaded.settings };
                this.state = {
                    ...this.state,
                    ...loaded,
                    settings,
                    ui: { ...this.state.ui, ...loaded.ui }
                };
            }
            
            this.state.exams.forEach(exam => { 
                if (!exam.id) exam.id = Date.now() + Math.random();
                // Initialize new state properties for older saved states
                exam.isPaused = exam.isPaused || false;
                exam.pauseStartTime = exam.pauseStartTime || null;
                exam.pauseDurationTotal = exam.pauseDurationTotal || 0;
                exam.status = exam.status || 'running';
            });
            
            if (this.state.isLive || this.state.isPaused) {
                this.state.currentPage = 'exam-page';
                this.prepareExamPage();
                this.startTimers();
                
                if(this.state.isPaused) {
                    this.showEmergencyModal(null, true); 
                }
            } else {
                this.state.isLive = false;
                this.state.isPaused = false;
                this.state.pauseStartTime = null;
                this.state.pauseDurationTotal = 0;
                this.state.autoStartTargetTime = null;
                this.state.currentPage = 'setup-page';
            }

            const savedTheme = localStorage.getItem('theme');
            document.documentElement.className = savedTheme || 'light';
            this.ui.darkModeToggle.checked = document.documentElement.classList.contains('dark');
            this.updateThemeIcons();
        },

        updateSettings() {
            const titleSelect = this.ui.inputs.sessionTitleSelect;
            this.state.settings.sessionTitle = titleSelect.value === 'Other' ? this.ui.inputs.sessionTitleOther.value : titleSelect.value;
            this.ui.inputs.sessionTitleOther.classList.toggle('hidden', titleSelect.value !== 'Other');
            
            // Toggle preset button
            this.ui.presetBtn.disabled = !this.examPresets[this.state.settings.sessionTitle];

            for (const key in this.defaultSettings) {
                const input = this.ui.inputs[`option${key.charAt(0).toUpperCase() + key.slice(1)}`];
                if (input) {
                    this.state.settings[key] = input.type === 'checkbox' ? input.checked : (input.tagName === 'SELECT' ? parseInt(input.value, 10) : input.value);
                }
            }
            
            const showTimes = this.state.settings.showTimes;
            this.ui.subOptionTimeBreakdown.classList.toggle('hidden', !showTimes);
            this.ui.subOptionSingleLine.classList.toggle('hidden', !showTimes);

            this.saveState();
        },

        render() {
            Object.values(this.ui.pages).forEach(page => page.classList.add('hidden'));
            this.ui.pages[this.state.currentPage.replace('-page', '')].classList.remove('hidden');

            this.ui.inputs.sessionTitleSelect.value = this.state.settings.sessionTitle;
            if (this.ui.inputs.sessionTitleSelect.selectedIndex === -1) {
                this.ui.inputs.sessionTitleSelect.value = 'Other';
                this.ui.inputs.sessionTitleOther.value = this.state.settings.sessionTitle;
            }
            this.ui.inputs.sessionTitleOther.classList.toggle('hidden', this.ui.inputs.sessionTitleSelect.value !== 'Other');
            
            this.ui.presetBtn.disabled = !this.examPresets[this.state.settings.sessionTitle];
            
            for (const key in this.defaultSettings) {
                const input = this.ui.inputs[`option${key.charAt(0).toUpperCase() + key.slice(1)}`];
                if (input) {
                    if (input.type === 'checkbox') {
                        input.checked = this.state.settings[key];
                    } else {
                        input.value = this.state.settings[key];
                    }
                }
            }

            const showTimes = this.state.settings.showTimes;
            this.ui.subOptionTimeBreakdown.classList.toggle('hidden', !showTimes);
            this.ui.subOptionSingleLine.classList.toggle('hidden', !showTimes);
            
            this.ui.tooltipToggle.checked = this.state.settings.showTooltips;
            this.toggleAllTooltips(this.state.settings.showTooltips);

            this.renderExamList();
            this.applyFontSizes();
        },

            this.ui.examList.innerHTML = '';
            this.ui.noExamsMessage.classList.toggle('hidden', this.state.exams.length > 0);

            this.ui.commenceBtn.disabled = this.state.exams.length === 0;
            this.ui.commenceBtn.classList.toggle('opacity-50', this.state.exams.length === 0);
            this.ui.commenceBtn.classList.toggle('cursor-not-allowed', this.state.exams.length === 0);

                        <p class="font-bold text-header">${exam.name}</p>
                        <p class="text-sm text-muted">Read: ${exam.readMins}m, Write: ${exam.writeHrs}h ${exam.writeMins}m</p>
                    </div>
                    <div class="flex space-x-2">
                        <button onclick="app.editExam('${exam.id}')" class="p-2 text-muted hover:text-indigo-600 transition" aria-label="Edit ${exam.name}">
                            <svg xmlns="http://www.w3.org/2000/svg" width="20" height="20" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"><path d="M17 3a2.828 2.828 0 1 1 4 4L7.5 20.5 2 22l1.5-5.5L17 3z"></path></svg>
                        </button>
                        <button onclick="app.showConfirmModal('deleteExam', '${exam.id}')" class="p-2 text-muted hover:text-red-600 transition" aria-label="Delete ${exam.name}">
                            <svg xmlns="http://www.w3.org/2000/svg" width="20" height="20" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"><polyline points="3 6 5 6 21 6"></polyline><path d="M19 6v14a2 2 0 0 1-2 2H7a2 2 0 0 1-2-2V6m3 0V4a2 2 0 0 1 2-2h4a2 2 0 0 1 2 2v2"></path><line x1="10" y1="11" x2="10" y2="17"></line><line x1="14" y1="11" x2="14" y2="17"></line></svg>
                        </button>
                    </div>

                this.ui.examList.appendChild(examCard);
            });
        },

        showPage(pageId) { this.state.currentPage = pageId; this.render(); },
        showSetupPage() { this.showPage('setup-page'); },

        showExamModal(examId = null) {
            this.clearExamModal();
            if (examId) {
                 const exam = this.state.exams.find(e => e.id == examId);
                 if (exam) {
                    this.state.isEditing = true; this.state.editingId = exam.id;
                    document.getElementById('modal-title').innerText = 'Edit Examination';
                    this.ui.inputs.examId.value = exam.id; this.ui.inputs.examName.value = exam.name;
                    this.ui.inputs.readMins.value = exam.readMins; this.ui.inputs.writeHrs.value = exam.writeHrs;
                    this.ui.inputs.writeMins.value = exam.writeMins;
                    this.ui.inputs.optionalInfo.value = exam.optionalInfo;
                 }
            } else {
                this.state.isEditing = false; 
                document.getElementById('modal-title').innerText = 'Add Examination';
                const sessionTitle = this.ui.inputs.sessionTitleSelect.value;
                if (sessionTitle.includes('HSC')) {
                    this.ui.inputs.readMins.value = 10;
                    this.ui.inputs.writeHrs.value = 3;
                    this.ui.inputs.writeMins.value = 0;
                } else if (sessionTitle.includes('Preliminary') || sessionTitle.includes('Exit')) {
                    this.ui.inputs.readMins.value = 5;
                    this.ui.inputs.writeHrs.value = 2;
                    this.ui.inputs.writeMins.value = 0;
                }
            }
            this.openModal(this.ui.modals.exam);
        },
        hideExamModal() { this.closeModal(this.ui.modals.exam); },
        clearExamModal() {
            this.ui.inputs.examId.value = ''; this.ui.inputs.examName.value = '';
            this.ui.inputs.readMins.value = '5'; this.ui.inputs.writeHrs.value = '0';
            this.ui.inputs.writeMins.value = '30'; this.ui.inputs.optionalInfo.value = '';
            this.ui.examModalError.textContent = '';
        },
        saveExam() {
            const name = this.ui.inputs.examName.value.trim();
            if (!name) { this.ui.examModalError.textContent = 'Please provide an examination name.'; return; }
            
            const readMins = parseInt(this.ui.inputs.readMins.value, 10) || 0;
            const writeHrs = parseInt(this.ui.inputs.writeHrs.value, 10) || 0;
            const writeMins = parseInt(this.ui.inputs.writeMins.value, 10) || 0;
            const optionalInfo = this.ui.inputs.optionalInfo.value.trim();

            if (this.state.isEditing) {
                const index = this.state.exams.findIndex(e => e.id === this.state.editingId);
                if (index > -1) {
                    const exam = this.state.exams[index];
                    exam.name = name; exam.readMins = readMins; exam.writeHrs = writeHrs;
                    exam.writeMins = writeMins; exam.optionalInfo = optionalInfo;
                    
                    if (this.state.isLive && exam.startTime) {
                        const readMillis = exam.readMins * 60 * 1000;
                        const writeMillis = (exam.writeHrs * 3600 + exam.writeMins * 60) * 1000;
                        exam.readEndTime = exam.startTime + readMillis;
                        exam.writeEndTime = exam.readEndTime + writeMillis;
                    }
                }
            } else {
                this.state.exams.push({ 
                    id: Date.now() + Math.random(), name, readMins, writeHrs, writeMins, optionalInfo,
                    isPaused: false, pauseStartTime: null, pauseDurationTotal: 0, status: 'running'
                });
            }
            
            this.saveState(); this.hideExamModal();
            if (this.state.currentPage === 'exam-page') {
                this.prepareExamPage();
            }
            else {
                this.renderExamList();
            }
        },
        editExam(id) {
            if (this.state.isLive) {
                this.showConfirmModal('editLiveExam', id);
            } else {
                this.showExamModal(id);
            }
        },

            this.hideConfirmModal();
            this.showExamModal(id);
        },

            const title = document.getElementById('confirm-title');
            const message = document.getElementById('confirm-message');
            const btn = this.ui.confirmActionBtn;
            
            const actions = {
                deleteExam: { title: 'Delete Exam?', msg: 'Are you sure you want to permanently delete this examination?', onclick: () => this.deleteExamAction(id) },
                clearAll: { title: 'Clear All Exams?', msg: 'This will delete all examinations you have added. This cannot be undone.', onclick: () => this.clearAllExamsAction() },
                resetAll: { title: 'Reset All Settings?', msg: 'This will restore all display options to their defaults and clear all exams.', onclick: () => this.resetAllAction() },

        hideConfirmModal() { this.closeModal(this.ui.modals.confirm); },
        deleteExamAction(id) {
            this.state.exams = this.state.exams.filter(e => e.id != id);
            this.saveState();
            this.hideConfirmModal();
            const elemToRemove = document.getElementById(`setup-exam-${id}`);
            if (elemToRemove) {
                elemToRemove.remove();
                 this.ui.noExamsMessage.classList.toggle('hidden', this.state.exams.length > 0);
            } else {
                this.renderExamList();
            }
        },
        clearAllExamsAction() {
            this.state.exams = [];
            this.saveState();
            this.hideConfirmModal();
            this.renderExamList();
        },
        resetAllAction() {
            this.state.settings = { ...this.defaultSettings };
            this.state.exams = [];
            this.saveState();
            this.hideConfirmModal();
            this.render();
        },
        confirmReturnToSetup() { this.showConfirmModal('returnToSetup'); },
        
        showAutoStartModal() { this.openModal(this.ui.modals.autoStart); },
        hideAutoStartModal() { this.closeModal(this.ui.modals.autoStart); },
        setAutoStart() {
            const timeValue = this.ui.inputs.autoStartTime.value;
            const errorEl = this.ui.autoStartError;
            if (!timeValue) { errorEl.textContent = 'Please select a valid time.'; return; }
            const [hours, minutes] = timeValue.split(':');
            const now = new Date();
            const startTime = new Date(now.getFullYear(), now.getMonth(), now.getDate(), hours, minutes, 0);
            if (startTime <= now) { errorEl.textContent = 'Start time must be in the future.'; return; }
            
            this.state.autoStartTargetTime = startTime.getTime();
            this.saveState();
            this.hideAutoStartModal();
            this.updateAutoStartBanner();
            this.prepareExamPage();

            this.intervals.autoStart = setInterval(() => {
                if (new Date() >= this.state.autoStartTargetTime) {
                    this.cancelAutoStart();
                    if(this.state.currentPage !== 'exam-page') this.previewExams();
                    this.startLiveExams();
                }
            }, 1000);
        },
        cancelAutoStart() {
            clearInterval(this.intervals.autoStart);
            this.intervals.autoStart = null;
            this.state.autoStartTargetTime = null;
            this.saveState();
            this.updateAutoStartBanner();
            this.prepareExamPage();
        },
        updateAutoStartBanner() {
            if (this.state.autoStartTargetTime) {
                const startTime = new Date(this.state.autoStartTargetTime);
                this.ui.autoStartBannerMessage.textContent = `Exams will automatically commence at ${this.formatClockTime(startTime)}.`;
                this.ui.autoStartBanner.classList.remove('hidden');
            } else {
                this.ui.autoStartBanner.classList.add('hidden');
            }
        },

        previewExams() {
            if (this.state.exams.length === 0) return;
            this.state.isLive = false; 
            this.showPage('exam-page');
            this.prepareExamPage(); 
            this.startTimers();
            this.initSortableExamPage();
            window.scrollTo(0, 0);
        },
        async startLiveExams() {
            try {
                if (document.documentElement.requestFullscreen) {
                    await document.documentElement.requestFullscreen();
                }
                if ('wakeLock' in navigator) {
                    this.wakeLockSentinel = await navigator.wakeLock.request('screen');
                }
            } catch (err) {
                console.error(`Failed to activate fullscreen or wake lock: ${err.name}, ${err.message}`);
            }
            window.scrollTo(0, 0);
            this.state.isLive = true;
            this.state.isPaused = false;
            this.state.pauseDurationTotal = 0;
            this.ui.previewActions.classList.add('hidden'); this.ui.liveActions.classList.remove('hidden');
            const now = Date.now();
            this.state.exams.forEach(exam => {
                const readMillis = exam.readMins * 60 * 1000;
                const writeMillis = (exam.writeHrs * 3600 + exam.writeMins * 60) * 1000;
                exam.startTime = now; exam.readEndTime = now + readMillis;
                exam.writeEndTime = exam.readEndTime + writeMillis;
            });
            this.updateExams(); this.saveState();
        },

        },

        startTimers() {
            Object.values(this.intervals).forEach(clearInterval);
            clearTimeout(this.intervals.mainClock);
            this.updateClocks();
            this.intervals.examUpdater = setInterval(() => this.updateExams(), 1000);
            this.updateExams();
        },

        updateClocks() {
            if (this.state.isPaused) return; // Global pause check for main clock
            const now = new Date();
            document.getElementById('header-date').textContent = now.toLocaleDateString(undefined, { weekday: 'long', year: 'numeric', month: 'long', day: 'numeric' });
            document.getElementById('header-time').textContent = this.formatClockTime(now);

            const msUntilNextSecond = 1000 - now.getMilliseconds();
            this.intervals.mainClock = setTimeout(() => this.updateClocks(), msUntilNextSecond);
        },

        updateExams() {
            const now = Date.now();
            this.state.exams.forEach(exam => {
                if (exam.status !== 'running') return; 
                if (this.state.isPaused || exam.isPaused) return;

                const card = document.getElementById(`exam-card-${exam.id}`);
                if (!card) return;
                let currentStatus = this.state.isLive ? (now < exam.readEndTime ? "Reading Time" : (now < exam.writeEndTime ? "Writing Time" : "Time Up")) : "Preview";
                
                if (currentStatus === "Time Up") {
                    exam.status = 'finished';
                    card.classList.add('is-finished');
                    card.classList.remove('reading-time', 'writing-time', 'last-10-mins');
                    this.saveState();
                }

                const statusEl = document.getElementById(`status-${exam.id}`);
                const timesEl = document.getElementById(`exam-times-${exam.id}`);
                
                if (this.state.settings.disableTimers) {
                    statusEl.innerHTML = '';
                    timesEl.innerHTML = '';
                    card.classList.remove('reading-time', 'writing-time', 'last-10-mins', 'is-finished');
                    return;
                }

                if (this.state.settings.showStatus) {
                    statusEl.innerHTML = currentStatus !== 'Time Up' ? currentStatus : `<span class="finished-text">Finished</span>`;
                } else {
                    statusEl.innerHTML = '';
                }

                if (this.state.settings.showTimes) {
                    let timesHTML = '';
                    if (this.state.isLive || this.state.autoStartTargetTime) {
                         if (this.state.settings.singleLineTime) {
                             if(this.state.settings.timeBreakdown) {
                                timesHTML = `<p><span class="font-semibold text-muted">R:</span> ${this.formatClockTimeShort(new Date(exam.startTime))}-${this.formatClockTimeShort(new Date(exam.readEndTime))} | <span class="font-semibold text-muted">W:</span> ${this.formatClockTimeShort(new Date(exam.readEndTime))}-${this.formatClockTimeShort(new Date(exam.writeEndTime))}</p>`;
                             } else {
                                timesHTML = `<p><span class="font-semibold text-muted">Start:</span> ${this.formatClockTimeShort(new Date(exam.startTime))} - <span class="font-semibold text-muted">End:</span> ${this.formatClockTimeShort(new Date(exam.writeEndTime))}</p>`;
                             }
                         } else {
                             if (this.state.settings.timeBreakdown) {
                                timesHTML = `
                                    <p><span class="font-semibold text-muted">Reading Start:</span> ${this.formatClockTimeShort(new Date(exam.startTime))}</p>
                                    <p><span class="font-semibold text-muted">Reading End:</span> ${this.formatClockTimeShort(new Date(exam.readEndTime))}</p>
                                    <p><span class="font-semibold text-muted">Writing Start:</span> ${this.formatClockTimeShort(new Date(exam.readEndTime))}</p>
                                    <p><span class="font-semibold text-muted">Writing End:</span> ${this.formatClockTimeShort(new Date(exam.writeEndTime))}</p>
                                `;
                            } else {
                                 timesHTML = `
                                    <p><span class="font-semibold text-muted">Start:</span> ${this.formatClockTimeShort(new Date(exam.startTime))}</p>
                                    <p><span class="font-semibold text-muted">End:</span> ${this.formatClockTimeShort(new Date(exam.writeEndTime))}</p>
                                `;
                            }
                         }
                    } else {
                        timesHTML = `
                            <p><span class="font-semibold text-muted">Start:</span> --:--</p>
                            <p><span class="font-semibold text-muted">End:</span> --:--</p>
                        `;
                    }
                    timesEl.innerHTML = timesHTML;
                } else {
                    timesEl.innerHTML = '';
                }

                if (this.state.settings.colorAlerts) {
                    card.classList.remove('reading-time', 'writing-time', 'last-10-mins');
                    if (currentStatus === "Reading Time") card.classList.add('reading-time');
                    else if (currentStatus === "Writing Time") {
                        if (exam.writeEndTime - now <= 10 * 60 * 1000) card.classList.add('last-10-mins');
                        else card.classList.add('writing-time');
                    }
                } else {
                     card.classList.remove('reading-time', 'writing-time', 'last-10-mins');
                }
            });
        },
        
        async returnToSetup() {
            Object.values(this.intervals).forEach(clearInterval);
            clearTimeout(this.intervals.mainClock);

            if (this.wakeLockSentinel) {
                await this.wakeLockSentinel.release();
                this.wakeLockSentinel = null;
            }
            if (document.fullscreenElement) {
                await document.exitFullscreen();
            }

            this.state.isLive = false; 
            this.state.isPaused = false;
            this.state.autoStartTargetTime = null;
            // Reset all exam statuses for a fresh start
            this.state.exams.forEach(exam => {
                exam.status = 'running';
                exam.isPaused = false;
                exam.pauseStartTime = null;
                exam.pauseDurationTotal = 0;
            });
            this.saveState();
            this.showSetupPage();
            this.hideConfirmModal();
        },

        showEmergencyModal(examId = null, isResuming = false) {
            this.ui.inputs.emergencyJustification.value = '';
            this.ui.emergencyError.textContent = '';
            const exam = examId ? this.state.exams.find(e => e.id == examId) : null;
            
            if (isResuming || (exam && exam.isPaused) || (!exam && this.state.isPaused)) {
                this.ui.emergencyContent.classList.add('hidden');
                this.ui.pausedContent.classList.remove('hidden');
                this.ui.pausedTitle.textContent = exam ? `EXAM PAUSED: ${exam.name}` : 'SESSION PAUSED';
                this.ui.pausedResumeBtn.onclick = () => this.resumeSession(examId);
            } else {
                this.ui.emergencyContent.classList.remove('hidden');
                this.ui.pausedContent.classList.add('hidden');
                this.ui.emergencyMessage.textContent = exam 
                    ? `A disruption has occurred for exam: ${exam.name}. Choose an action below.`
                    : 'A disruption has occurred for the entire session. Choose an action below.';
                this.ui.emergencyPauseBtn.onclick = () => this.pauseSession(examId);
                this.ui.emergencyAbandonBtn.onclick = () => this.showConfirmModal('abandon', examId);
            }
            this.openModal(this.ui.modals.emergency);
        },
        hideEmergencyModal() { this.closeModal(this.ui.modals.emergency); },
        pauseSession(examId = null) {
            const justification = this.ui.inputs.emergencyJustification.value.trim();
            if (!justification) { this.ui.emergencyError.textContent = "Please provide a justification for pausing."; return; }

            const exam = examId ? this.state.exams.find(e => e.id == examId) : null;
            if (exam) { // Individual pause
                exam.isPaused = true;
                exam.pauseStartTime = Date.now();
                document.getElementById(`exam-paused-overlay-${exam.id}`).classList.remove('hidden');
            } else { // Global pause
                this.state.isPaused = true;
                this.state.pauseStartTime = Date.now();
                this.state.exams.forEach(e => {
                    if (!e.isPaused) document.getElementById(`exam-paused-overlay-${e.id}`).classList.remove('hidden');
                });
            }

            this.exportSessionLog(exam ? `Paused Exam: ${exam.name}` : "Paused Session", justification);
            this.hideEmergencyModal(); // Close the modal for in-card resume
            this.saveState();
        },
        resumeSession(examId = null) {
            const exam = examId ? this.state.exams.find(e => e.id == examId) : null;

            if (exam) { // Individual resume
                const pauseDuration = Date.now() - exam.pauseStartTime;
                exam.pauseDurationTotal += pauseDuration;
                exam.readEndTime += pauseDuration;
                exam.writeEndTime += pauseDuration;
                exam.isPaused = false;
                exam.pauseStartTime = null;
                document.getElementById(`exam-paused-overlay-${exam.id}`).classList.add('hidden');
                this.exportSessionLog(`Resumed Exam: ${exam.name}`, "Resumed by supervisor.");
            } else { // Global resume
                const pauseDuration = Date.now() - this.state.pauseStartTime;
                this.state.pauseDurationTotal += pauseDuration;
                this.state.exams.forEach(e => {
                    if (!e.isPaused) {
                         e.readEndTime += pauseDuration;
                         e.writeEndTime += pauseDuration;
                         document.getElementById(`exam-paused-overlay-${e.id}`).classList.add('hidden');
                    }
                });
                this.state.isPaused = false;
                this.state.pauseStartTime = null;
                this.hideEmergencyModal();
                this.exportSessionLog("Resumed Session", "Resumed by supervisor.");
            }

            this.saveState();
            this.updateClocks();
        },
        abandonSessionAction(examId = null) {
            const justification = this.ui.inputs.emergencyJustification.value.trim();
            if (!justification) { 
                this.hideConfirmModal();
                this.ui.emergencyError.textContent = "Please provide a justification for abandoning.";
                return; 
            }

            const exam = examId ? this.state.exams.find(e => e.id == examId) : null;
            if (exam) {
                exam.status = 'abandoned';
                const card = document.getElementById(`exam-card-${exam.id}`);
                card.classList.add('is-abandoned');
                document.getElementById(`status-${exam.id}`).innerHTML = ``;
                this.exportSessionLog(`Abandoned Exam: ${exam.name}`, justification);
                this.hideConfirmModal();
                this.hideEmergencyModal();
            } else {
                 this.exportSessionLog("Abandoned Session", justification);
                 this.hideConfirmModal();
                 this.hideEmergencyModal();
                 this.returnToSetup();
            }
            this.saveState();
        },
        exportSessionLog(status, justification) {
            let log = `EXAMINATION SESSION LOG\n=========================\n`;
            log += `Status: ${status}\nTime of Event: ${new Date().toLocaleString('en-AU')}\nJustification: ${justification}\n\n`;
            log += `SESSION DETAILS\n-------------------------\n`;
            log += `Title: ${this.state.settings.sessionTitle}\n`;
            if(this.state.settings.showSchool) log += `School: Inverell High School\n`;
            if(this.state.settings.showCentre) log += `Centre: Examination Centre 222\n\n`;
            log += `EXAMS IN PROGRESS\n-------------------------\n`;
            this.state.exams.forEach(exam => {
                const globalPause = this.state.isPaused ? (Date.now() - this.state.pauseStartTime) : 0;
                const totalPause = exam.pauseDurationTotal + (exam.isPaused ? (Date.now() - exam.pauseStartTime) : 0) + globalPause;
                const originalStartTime = new Date(exam.startTime - totalPause);
                const originalEndTime = new Date(exam.writeEndTime - totalPause);
                let currentStatus = exam.status;
                if(exam.isPaused) currentStatus = "paused";
                log += `\nExam: ${exam.name} (${currentStatus})\n`;
                log += `  - Original Start: ${originalStartTime.toLocaleString('en-AU')}\n`;
                log += `  - Original End: ${originalEndTime.toLocaleString('en-AU')}\n`;
                log += `  - Adjusted End: ${new Date(exam.writeEndTime).toLocaleString('en-AU')}\n`;
            });
            const blob = new Blob([log], { type: 'text/plain' });
            const link = document.createElement('a');
            link.href = URL.createObjectURL(blob);
            link.download = `Exam_Log_${new Date().toISOString().slice(0,10)}.txt`;
            link.click();
        },

        // Modal Logic
        showWelcomeModal() { this.openModal(this.ui.modals.welcome); },
        hideWelcomeModal() { 
            this.closeModal(this.ui.modals.welcome); 
            sessionStorage.setItem('welcomeShown', 'true');
        },
        showPresetModal() {
            const presetKey = this.state.settings.sessionTitle;
            const presets = this.examPresets[presetKey] || {};
            this.ui.presetExamList.innerHTML = '';

            for (const kla in presets) {
                const klaHeading = document.createElement('h3');
                klaHeading.className = 'kla-heading col-span-1 sm:col-span-2';
                klaHeading.textContent = kla;
                this.ui.presetExamList.appendChild(klaHeading);

                const listContainer = document.createElement('div');
                listContainer.className = "col-span-1 sm:col-span-2 grid grid-cols-1 sm:grid-cols-2 gap-x-4 gap-y-2";

                presets[kla].forEach((preset, index) => {
                    const div = document.createElement('div');
                    div.className = 'flex items-center';
                    const uniqueId = `preset-${kla}-${index}`;
                    div.innerHTML = `
                        <input id="${uniqueId}" type="checkbox" value="${preset.name}" class="h-4 w-4 text-indigo-600 border-gray-300 rounded focus:ring-indigo-500">
                        <label for="${uniqueId}" class="ml-3 block text-sm text-body">${preset.name}</label>
                    `;
                    listContainer.appendChild(div);
                });
                this.ui.presetExamList.appendChild(listContainer);
            }
            this.openModal(this.ui.modals.preset);
        },
        hidePresetModal() { this.closeModal(this.ui.modals.preset); },
        calculateExamTimes(totalMins) {
            // Try 10 mins reading first
            let readMins = 10;
            let writeTotalMins = totalMins - readMins;

            // If writing time is not a multiple of 30, try 5 mins reading
            if (writeTotalMins % 30 !== 0) {
                readMins = 5;
                writeTotalMins = totalMins - readMins;
            }
            
            // If it's still not right (for short exams), default to 5.
            if (writeTotalMins <= 0) {
                readMins = 5;
                writeTotalMins = totalMins - readMins;
            }

            const writeHrs = Math.floor(writeTotalMins / 60);
            const writeMins = writeTotalMins % 60;
            
            return { readMins, writeHrs, writeMins };
        },
        addSelectedPresets() {
            const selectedExams = Array.from(this.ui.presetExamList.querySelectorAll('input[type="checkbox"]:checked'));
            const presetKey = this.state.settings.sessionTitle;
            const presetsByKla = this.examPresets[presetKey] || {};
            
            const allPresets = Object.values(presetsByKla).flat();

            selectedExams.forEach(checkbox => {
                const examName = checkbox.value;
                const preset = allPresets.find(p => p.name === examName);
                if (preset && !this.state.exams.some(e => e.name === preset.name)) {
                    const { readMins, writeHrs, writeMins } = this.calculateExamTimes(preset.total);
                    this.state.exams.push({
                        id: Date.now() + Math.random(),
                        name: preset.name,
                        readMins, writeHrs, writeMins,
                        optionalInfo: '',
                        isPaused: false, pauseStartTime: null, pauseDurationTotal: 0, status: 'running'
                    });
                }
            });
            this.hidePresetModal();
            this.renderExamList();
            this.saveState();
        },
        showLiveSettingsModal() {
            this.ui.inputs.liveGridLayout.value = this.state.settings.gridLayout;
            this.ui.inputs.liveShowCrest.checked = this.state.settings.showCrest;
            this.ui.inputs.liveColorAlerts.checked = this.state.settings.colorAlerts;
            this.ui.inputs.liveShowStatus.checked = this.state.settings.showStatus;
            this.ui.inputs.liveShowSeconds.checked = this.state.settings.showSeconds;
            this.ui.inputs.liveSingleLineTime.checked = this.state.settings.singleLineTime;
            this.openModal(this.ui.modals.liveSettings);
        },
        applyLiveSettings() {
            this.state.settings.gridLayout = parseInt(this.ui.inputs.liveGridLayout.value, 10);
            this.state.settings.showCrest = this.ui.inputs.liveShowCrest.checked;
            this.state.settings.colorAlerts = this.ui.inputs.liveColorAlerts.checked;
            this.state.settings.showStatus = this.ui.inputs.liveShowStatus.checked;
            this.state.settings.showSeconds = this.ui.inputs.liveShowSeconds.checked;
            this.state.settings.singleLineTime = this.ui.inputs.liveSingleLineTime.checked;
            this.saveState();
            this.closeModal(this.ui.modals.liveSettings);
            this.prepareExamPage();
        },
        openModal(modalElement) {
            this.modalFocus.previouslyFocusedEl = document.activeElement;
            this.modalFocus.activeModal = modalElement;
            const focusable = modalElement.querySelectorAll('button, [href], input, select, textarea, [tabindex]:not([tabindex="-1"])');
            this.modalFocus.focusableEls = Array.from(focusable);
            this.modalFocus.firstFocusableEl = this.modalFocus.focusableEls[0];
            this.modalFocus.lastFocusableEl = this.modalFocus.focusableEls[this.modalFocus.focusableEls.length - 1];
            modalElement.classList.remove('hidden');
            this.modalFocus.firstFocusableEl?.focus();
        },
        closeModal(modalElement) {
            modalElement.classList.add('hidden');
            this.modalFocus.previouslyFocusedEl?.focus();
            this.modalFocus.activeModal = null;
        },
        handleModalFocus(e) {
            if (!this.modalFocus.activeModal || e.key !== 'Tab') return;
            
            if (e.shiftKey) {
                if (document.activeElement === this.modalFocus.firstFocusableEl) {
                    this.modalFocus.lastFocusableEl.focus();
                    e.preventDefault();
                }
            } else {
                if (document.activeElement === this.modalFocus.lastFocusableEl) {
                    this.modalFocus.firstFocusableEl.focus();
                    e.preventDefault();
                }
            }
        },

        toggleFabs() {
            this.state.ui.fabsCollapsed = !this.state.ui.fabsCollapsed;
            this.renderFabs();
            this.saveState();
        },
        renderFabs() {
            const container = this.ui.fabActionsContainer;
            const controls = this.ui.fabControlsContainer;
            if (this.state.ui.fabsCollapsed) {
                container.classList.add('hidden');
                controls.classList.add('hidden');
                this.ui.fabToggleIcon.classList.add('rotate-180');
            } else {
                container.classList.remove('hidden');
                controls.classList.remove('hidden');
                this.ui.fabToggleIcon.classList.remove('rotate-180');
            }
        },

        // Utilities
        formatClockTime(date) { 
            const options = { 
                hour: this.state.settings.is24hr ? '2-digit' : 'numeric', 
                minute: '2-digit', 
                hour12: !this.state.settings.is24hr 
            };
            if(this.state.settings.showSeconds) {
                options.second = '2-digit';
            }
            return date.toLocaleTimeString('en-AU', options);
        },
        formatClockTimeShort(date) { return date.toLocaleTimeString('en-AU', { hour: this.state.settings.is24hr ? '2-digit' : 'numeric', minute: '2-digit', hour12: !this.state.settings.is24hr }); },
        toggleDarkMode() {
            document.documentElement.classList.toggle('dark');
            localStorage.setItem('theme', document.documentElement.classList.contains('dark') ? 'dark' : 'light');
            this.updateThemeIcons();
        },
        updateThemeIcons() {
            const isDark = document.documentElement.classList.contains('dark');
            this.ui.sunIcon.classList.toggle('hidden', isDark);
            this.ui.moonIcon.classList.toggle('hidden', !isDark);
        },
        toggleFontControls() { this.ui.appContainer.classList.toggle('show-font-controls'); },
        toggleAllTooltips(show) { this.ui.body.classList.toggle('tooltips-enabled', show); },
        positionTooltip(event) {
            const tooltipContent = event.currentTarget.querySelector('.tooltip-content');
            if (!tooltipContent) return;

            // Use a microtask to ensure the element is rendered and has dimensions
            setTimeout(() => {
                const rect = event.currentTarget.getBoundingClientRect();
                const contentRect = tooltipContent.getBoundingClientRect();
                const viewportWidth = window.innerWidth;
                
                // Reset styles to recalculate position
                tooltipContent.style.left = '50%';
                tooltipContent.style.transform = 'translateX(-50%)';
                
                if (contentRect.left < 10) {
                    tooltipContent.style.left = `${-rect.left + 10}px`;
                    tooltipContent.style.transform = 'translateX(0)';
                } else if (contentRect.right > viewportWidth - 10) {
                     tooltipContent.style.left = 'auto';
                     tooltipContent.style.right = `${viewportWidth - rect.right - 10}px`;
                     tooltipContent.style.transform = 'translateX(0)';
                }
            }, 0);
        },
        toggleFontLock() {
            this.state.ui.fontLockEnabled = !this.state.ui.fontLockEnabled;
            this.ui.fontLockToggle.classList.toggle('locked', this.state.ui.fontLockEnabled);
            this.saveState();
        },
        changeFontSize(elementId, direction) {
            const change = (id) => {
                if (!this.state.settings.fontSizes[id]) {
                    const el = document.getElementById(id);
                    if(el) this.state.settings.fontSizes[id] = parseInt(window.getComputedStyle(el).fontSize);
                    else return;
                }
                const currentSize = this.state.settings.fontSizes[id];
                const newSize = direction === 'up' ? currentSize + 2 : currentSize - 2;
                this.state.settings.fontSizes[id] = Math.max(12, newSize);
            };
            
            if (this.state.ui.fontLockEnabled) {
                 const idParts = elementId.split('-');
                 let typeAndPart = idParts[0];
                 if(idParts.length > 1) {
                    typeAndPart = (idParts[0] === 'exam') ? `${idParts[0]}-${idParts[1]}` : idParts[0];
                 }
                
                document.querySelectorAll(`[id^="${typeAndPart}-"]`).forEach(el => {
                    change(el.id);
                });
            } else {
                change(elementId);
            }
            this.applyFontSizes();
            this.saveState();
        },
        applyFontSizes() {
            for (const id in this.state.settings.fontSizes) {
                const el = document.getElementById(id);
                if (el) el.style.fontSize = `${this.state.settings.fontSizes[id]}px`;
            }
        },
    };

    window.app = app;
    document.addEventListener('DOMContentLoaded', () => app.init());
})();
